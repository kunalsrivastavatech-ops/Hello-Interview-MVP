    <script>
        function validateAndLaunch() {
            // 1. Data nodes ko capture karo
            const rollNumber = document.getElementById('student-roll').value.trim();
            const targetTrack = document.getElementById('track-select').value;
            const mediaChecked = document.getElementById('check-media').checked;
            const lockChecked = document.getElementById('check-lock').checked;

            // 2. C++ style validation checks logic loop
            if (rollNumber === "") {
                alert("❌ Input Error: Please enter your valid GLA Student Roll Number.");
                return;
            }
            if (!mediaChecked || !lockChecked) {
                alert("❌ Verification Failed: You must accept all system and compliance checks to run the simulation.");
                return;
            }

            // 3. The Local Storage Secret Hack
            // Yeh commands user ke parameters ko browser ki temporary cache memory me lock kar dete hain
            localStorage.setItem('candidateRoll', rollNumber);
            localStorage.setItem('selectedTrack', targetTrack);

            alert("🚀 System Validated successfully!\nSaving cache parameters...\n\nInitializing proctoring graphics and opening your live testing arena!");
            
            // Redirecting user action chain to the next page layout file
            window.location.href = 'arena.html';
        }
    </script>
